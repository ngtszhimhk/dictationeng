function takeScreenshot() { const element = document.getElementById('screenshotArea'); const btn = document.getElementById('btnScreenshot'); const originalText = btn.innerText; btn.innerText = '📸 正在處理成績單...'; btn.disabled = true; html2canvas(element, { backgroundColor: '#ffffff', scale: 2 }).then(canvas => { // 將畫布轉換為 Blob (檔案格式)，以便給手機的分享功能使用 canvas.toBlob(async (blob) => { const fileName = '默書成績單.png'; const file = new File([blob], fileName, { type: 'image/png' }); // 檢查裝置是否支援「系統分享」功能 (通常手機、平板都有支援) if (navigator.canShare && navigator.canShare({ files: [file] })) { try { await navigator.share({ files: [file], title: '我的默書成績單', text: '這是我剛剛完成的英文默書成績！' }); // 分享或儲存成功後恢復按鈕狀態 btn.innerText = originalText; btn.disabled = false; } catch (err) { // 如果學生按了「取消」，或者分享失敗，就退回傳統下載模式 console.log("分享取消或失敗，改用傳統下載", err); forceDownload(canvas.toDataURL('image/png'), fileName); } } else { // 如果是電腦版或不支援分享的瀏覽器，直接用傳統下載 forceDownload(canvas.toDataURL('image/png'), fileName); } }, 'image/png'); }).catch(err => { alert("截圖發生錯誤：" + err); btn.innerText = originalText; btn.disabled = false; }); // 傳統下載功能 (獨立寫成一個小函數方便呼叫) function forceDownload(dataUrl, filename) { const link = document.createElement('a'); link.download = filename; link.href = dataUrl; link.click(); btn.innerText = originalText; btn.disabled = false; } }
